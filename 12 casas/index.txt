const imagenesCaballeros = {
    'Cisne': 'Cisne.png', 'Dragón': 'Dragon.png', 'Pegaso': 'Pegaso.png', 'Fénix': 'Fenix.png', 'Andrómeda': 'Andromeda.png'

let casasSantuario = [], currentHouseIdx = 0, currentQuestionIdx = 0;
let lives = 5, maxLives = 5, score = 0, chosenKnight = "", playerName = "";
let bossHP = 0, bossMaxHP = 0;
let tieneHabilidadEspecial = false; // Variable de control para salvarse en la Casa 13

window.onload = function() {
    const saved = localStorage.getItem('casasSantuario');
    if(saved) {
        try { casasSantuario = JSON.parse(saved); } catch(e) { localStorage.removeItem('casasSantuario'); }
    }
};

function enterStudentMode() {
    if (casasSantuario && casasSantuario.length > 0) {
        document.getElementById('screen-portal').style.display = 'none';
        document.getElementById('screen-selection').style.display = 'flex';
    } else { alert("⚠️ El docente aún no ha cargado el archivo de preguntas (.txt)."); }
}

function showTeacherPasswordForm() {

function verifyTeacherPassword() {
    if(document.getElementById('teacherPassword').value === "SantaRosa123") {
        document.getElementById('portal-teacher-auth').style.display = 'none';
        document.getElementById('portal-teacher-upload').style.display = 'block';
    } else { alert("❌ Contraseña incorrecta del Santuario."); }
}

function processQuestionsFile() {
    const file = document.getElementById('questionsLoader').files[0];
    if(!file) return alert("Por favor, selecciona un archivo.");
    const reader = new FileReader();
    reader.onload = function(e) {
        casasSantuario = parseTextToQuestions(e.target.result);
        if(casasSantuario.length > 0) {
            localStorage.setItem('casasSantuario', JSON.stringify(casasSantuario));
            alert(`¡Éxito! Se cargaron ${casasSantuario.length} templos en la base de datos del Santuario.`);
            location.reload();
        } else { alert("Formato de archivo incorrecto."); }
    };
    reader.readAsText(file);
}

function parseTextToQuestions(text) {
    const lines = text.split('\n');
    const structure = [];
    let currentHouse = null, currentQuestion = null;
    const staticMap = jefesDatosEstaticos.reduce((map, item) => { map[item.h.toLowerCase()] = item; return map; }, {});

    lines.forEach(line => {
        line = line.trim();
        if (line.startsWith('#')) {
            const data = staticMap[line.substring(1).trim().toLowerCase()];
            if (data) {
                currentHouse = { house: data.h, boss: data.b, bossImg: data.img, questions: [] };
                structure.push(currentHouse);
            }
        } else if (currentHouse && line.toUpperCase().startsWith('PREGUNTA:')) {
            currentQuestion = { q: line.substring(9).trim(), options: [], a: -1 };
            currentHouse.questions.push(currentQuestion);
        } else if (currentQuestion && /^[A-E]\)/.test(line)) {
            let opt = line.substring(2).trim();
            if (opt.toUpperCase().includes('(CORRECTA)')) {
                currentQuestion.a = currentQuestion.options.length;
                opt = opt.replace(/\(CORRECTA\)/i, '').trim();
            }
            currentQuestion.options.push(opt);
        }
    });
    return structure;
}

    document.querySelectorAll('.knight-card').forEach(c => c.classList.remove('selected'));

    preview.src = imagenesCaballeros[name];

}

    const videoIntro = document.getElementById('intro-video');
    if (videoIntro) videoIntro.pause();

    document.getElementById('hud').style.display = 'flex';

    document.getElementById('player-arena-name').innerText = playerName;
    
    // Simulación: Otorgar la habilidad especial de forma predeterminada para pruebas
    tieneHabilidadEspecial = true; 

    handleSantuarioProgression();
}

function handleSantuarioProgression() {
    if(currentHouseIdx >= casasSantuario.length) {
        alert("🎉 ¡Felicidades! Has superado los 13 templos divinos y has salvado el universo."); 
        localStorage.removeItem('casasSantuario');
        return location.reload();
    }
    
    document.querySelectorAll('.screen').forEach(s => s.style.display = 'none');
    document.getElementById('screen-house-defense').style.display = 'flex';
    
    document.getElementById('defense-status-text').innerText = 
        (currentHouseIdx === 12) ? "⚠️ ¡Has entrado al Templo Secreto del Patriarca (Muerte Súbita)!" : "El destino del Santuario te espera...";
}

function enterHouseBattleFromDefense() {
    document.getElementById('screen-house-defense').style.display = 'none';
    document.getElementById('screen-battle').style.display = 'flex';
    startHouseCombat();
}

function startHouseCombat() {
    const house = casasSantuario[currentHouseIdx];
    document.getElementById('hud-house').innerText = house.house.toUpperCase();
    document.getElementById('boss-img').src = house.bossImg;
    document.getElementById('boss-arena-name').innerText = house.boss;
    
    currentQuestionIdx = 0;
    bossMaxHP = house.questions.length;
    bossHP = bossMaxHP;
    
    updateUI();
    loadQuestion();
}

function loadQuestion() {
    document.getElementById('feedback').innerText = "";
    const house = casasSantuario[currentHouseIdx];
    const q = house.questions[currentQuestionIdx];
    
    document.getElementById('question-text').innerText = `🔹 Desafío: ${q.q}`;
    const container = document.getElementById('options-container');
    container.innerHTML = '';
    
    q.options.forEach((opt, i) => {
        const btn = document.createElement('button');
        btn.className = 'option-btn';
        btn.innerText = opt;
        btn.onclick = () => checkAnswer(i, q.a);
        container.appendChild(btn);
    });
}

function checkAnswer(selected, correct) {
    // Verificar si estamos en la casa 13 (índice 12 del array de preguntas)
    const esCasa13 = (currentHouseIdx === 12);

    if(selected === correct) {
        bossHP--;
        score += 10;
        document.getElementById('feedback').innerHTML = "<span style='color:var(--hp-green)'>💥 ¡Cosmos eficaz! Impacto directo al guardián.</span>";
        updateUI();
        
        setTimeout(() => {
            currentQuestionIdx++;
            if(currentQuestionIdx >= casasSantuario[currentHouseIdx].questions.length) {
                currentHouseIdx++;
                handleSantuarioProgression();
            } else { loadQuestion(); }
        }, 1200);
    } else {
        if (esCasa13) {
            // Regla estricta Casa 13: Muerte súbita
            if (tieneHabilidadEspecial) {
                tieneHabilidadEspecial = false; // Se consume el escudo protector
                alert("⚡ ¡Tu Habilidad Especial te ha salvado del ataque mortal del Patriarca! Continúas con una oportunidad más.");
                document.getElementById('feedback').innerHTML = "<span style='color:orange'>🛡️ Habilidad Especial Consumida.</span>";
            } else {
                alert("💀 ¡Muerte Súbita en la Casa 13! Has fallado el golpe definitivo del Patriarca. Juego Terminado.");
                return location.reload();
            }
        } else {
            // Lógica normal para las primeras 12 casas
            lives--;
            updateUI();
            document.getElementById('feedback').innerHTML = "<span style='color:var(--danger)'>⚡ ¡El guardián bloquea tu ataque y te hiere!</span>";
            
            if(lives <= 0) {
                alert("💀 Tu energía vital se ha extinguido. Vuelve a intentarlo desde el primer templo."); 
                return location.reload();
            }
        }
    }
}

function updateUI() {
    const playerPerc = (lives / maxLives) * 100;
    document.getElementById('player-hp-bar').style.width = playerPerc + "%";
    
    const bossPerc = bossMaxHP === 0 ? 0 : (bossHP / bossMaxHP) * 100;
    document.getElementById('boss-hp-bar').style.width = bossPerc + "%";
    document.getElementById('hud-score').innerText = score;
}
